interface Bank {

       void getDetails();

       void deposit();

       void withdraw();

       void displayBalance();

}

class Account implements Bank {

         int accno;

       String name;

       double balance;

       Scanner sc = new Scanner(System.in);

       public void getDetails() {

               System.out.println("Enter account number:");

               accno=sc.nextInt();

               sc.nextLine();

               System.out.println("Enter account holder name:");

              name=sc.nextLine();

               System.out.println("Enter balance:");

              balance=sc.nextDouble();
              if (balance < 0) {
                  System.out.println("Initial balance cannot be negative");
                  balance = 0;
              }

      }

       public void deposit() {

             double amount;

              System.out.print("Enter amount to deposit: ");

             amount=sc.nextDouble();

             if (amount <= 0) {
                  System.out.println("Deposit amount must be positive");
                  return;
              }
              balance+=amount;

              System.out.println("Amount deposited successfully");
      }

       public void withdraw() {

             double amount;

               System.out.println("Enter amount to withdraw:");

             amount=sc.nextDouble();

             if (amount <= 0) {
                  System.out.println("Withdrawal amount must be positive");
              } else if(amount<=balance) {

                    balance-=amount;

                     System.out.println("Amount withdrawn successfully");

            }

               else {

                       System.out.println("Insufficient balance");

            }

      }

       public void displayBalance() {

               System.out.println("\n--- Amount Details ---");

              System.out.println("Account number: "+accno);

              System.out.println("Account Holder: "+name);

              System.out.println("Balance: "+balance);

      }

}

       public static void main(String[] args) {

            Account obj=new Account();

               System.out.println("--- Bank Account ---");

import java.util.Scanner;

abstract class shape {

        abstract void area();

       void display() {

               System.out.println("This is a shape");

      }

}



class Circle extends shape {

       double radius;

       Circle(double r) {

               radius = r;

      }

       void area() {

             double result = 3.14 * radius * radius;

              System.out.println("Area = " + result);

      }

       void circumference() {

             double result = 2 * 3.14 * radius;

              System.out.println("circumference = " + result);

      }

}



class Rectangle extends shape {

       double length, breadth;

       Rectangle(double l, double b) {

              length = l;
              breadth = b;

      }

       void area() {

             double result = length * breadth;

              System.out.println("Area = " + result);

      }

       void perimeter() {

             double result = 2 * (length + breadth);

               System.out.println("perimeter = " + result);

      }

}



public class MainAbstract {

       public static void main(String args[]) {

             Scanner sc = new Scanner(System.in);

              System.out.print("Enter the radius of circle: ");

             double r = sc.nextDouble();
              if (r < 0) {
                  System.out.println("Radius cannot be negative");
                  return;
              }

               Circle c = new Circle(r);



               System.out.println("\n--- Circle ---");

                c.display();

                c.area();

               c.circumference();



              System.out.print("Enter length: ");

             double l = sc.nextDouble();

              System.out.print("Enter breadth: ");

             double b = sc.nextDouble();
              if (l < 0 || b < 0) {
                  System.out.println("Rectangle dimensions cannot be negative");
                  return;
              }

SimpleCalculator.java
import javafx.application.Application;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.TextField;
import javafx.scene.layout.GridPane;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class SimpleCalculator extends Application {
    private TextField textField;
    private double firstNumber;
    private String operator = "";

    @Override
    public void start(Stage stage) {
        stage.setTitle("Simple Calculator");
        textField = new TextField();
        textField.setAlignment(Pos.CENTER_RIGHT);
        textField.setEditable(false);
        textField.setPrefWidth(260);

        String[] labels = {
            "7", "8", "9", "/",
            "4", "5", "6", "*",
            "1", "2", "3", "-",
            "C", "0", "=", "+",
            "%", "sin", "cos", "tan",
            "^"
        };
        GridPane grid = new GridPane();
        grid.setAlignment(Pos.CENTER);
        grid.setHgap(8);
        grid.setVgap(8);
        for (int i = 0; i < labels.length; i++) {
            Button button = new Button(labels[i]);
            button.setMinSize(54, 46);
            button.setOnAction(event -> handleButton(button.getText()));
            grid.add(button, i % 4, i / 4);
        }

        VBox root = new VBox(12, textField, grid);
        root.setAlignment(Pos.CENTER);
        root.setStyle("-fx-padding: 18;");
        stage.setScene(new Scene(root, 300, 390));
        stage.setResizable(false);
        stage.show();
    }

    private void handleButton(String command) {
        String current = textField.getText();
        if (command.matches("[0-9]")) {
            if (current.equals("Error")) current = "";
            textField.setText(current + command);
        } else if (command.equals("C")) {
            textField.clear();
            firstNumber = 0;
            operator = "";
        } else if (command.matches("sin|cos|tan")) {
            if (current.isEmpty() || current.equals("Error")) return;
            double degrees = Double.parseDouble(current);
            double radians = Math.toRadians(degrees);
            double answer;
            if (command.equals("sin")) answer = Math.sin(radians);
            else if (command.equals("cos")) answer = Math.cos(radians);
            else {
                if (Math.abs(Math.cos(radians)) < 1e-10) {
                    textField.setText("Error");
                    return;
                }
                answer = Math.tan(radians);
            }
            textField.setText(format(answer));
        } else if (command.matches("[+*/%\\-^]")) {
            if (current.isEmpty() || current.equals("Error")) return;
            firstNumber = Double.parseDouble(current);
            operator = command;
            textField.clear();
        } else if (command.equals("=")) {
            if (current.isEmpty() || operator.isEmpty() || current.equals("Error")) return;
            double secondNumber = Double.parseDouble(current);
            double result;
            switch (operator) {
                case "+": result = firstNumber + secondNumber; break;
                case "-": result = firstNumber - secondNumber; break;
                case "*": result = firstNumber * secondNumber; break;
                case "/":
                    if (secondNumber == 0) { textField.setText("Error"); operator = ""; return; }
                    result = firstNumber / secondNumber;
                    break;
                case "%":
                    if (secondNumber == 0) { textField.setText("Error"); operator = ""; return; }
                    result = firstNumber % secondNumber;
                    break;
                case "^": result = Math.pow(firstNumber, secondNumber); break;
                default: return;
            }
            textField.setText(Double.isFinite(result) ? format(result) : "Error");
            operator = "";
        }
    }

    private String format(double value) {
        if (Math.abs(value) < 1e-10) value = 0;
        return Double.toString(value);
    }

    public static void main(String[] args) {
        launch(args);
    }
}
